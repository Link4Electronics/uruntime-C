cmake_minimum_required(VERSION 3.25)
project(uruntime VERSION 0.6.0 LANGUAGES C
    DESCRIPTION "Universal RunImage and AppImage runtime with SquashFS and DwarFS supports"
)

set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(URUNTIME_APPIMAGE   "AppImage mode (vs RunImage)"                              OFF)
option(URUNTIME_SQUASHFS   "SquashFS support"                                         ON)
option(URUNTIME_DWARFS     "DwarFS support"                                           ON)
option(URUNTIME_LITE       "Lite mode (no mksquashfs/dwarfsck/mkdwarfs)"             OFF)
option(URUNTIME_UPX        "Embedded tools use UPX compression"                      OFF)
option(URUNTIME_STATIC     "Statically link all dependencies"                         ON)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Os -fvisibility=hidden)
if(URUNTIME_STATIC)
    add_compile_options(-ffunction-sections -fdata-sections)
    add_link_options(-Wl,--gc-sections -Wl,-z,noexecstack)
    # LTO only with static linking (avoids fat LTO objects for partial static)
    if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
    endif()
endif()
endif()

# Find dependencies via pkg-config
find_package(PkgConfig REQUIRED)
pkg_check_modules(ZSTD  REQUIRED libzstd)
pkg_check_modules(XXHASH QUIET libxxhash)

if(URUNTIME_STATIC)
    set(_saved ${CMAKE_FIND_LIBRARY_SUFFIXES})
    set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
    find_library(ZSTD_STATIC  NAMES zstd)
    find_library(XXHASH_STATIC NAMES xxhash)
    set(CMAKE_FIND_LIBRARY_SUFFIXES ${_saved})
endif()

# Build-time host tool - link shared (runs during build, not shipped)
add_executable(embed_compress EXCLUDE_FROM_ALL
    "${CMAKE_CURRENT_SOURCE_DIR}/embed_compress.c"
)
target_link_libraries(embed_compress PRIVATE ${ZSTD_LIBRARIES})
target_include_directories(embed_compress PRIVATE ${ZSTD_INCLUDE_DIRS})

# Select libraries for the target binary
set(URUNTIME_ZSTD_LIBS   ${ZSTD_LIBRARIES})
set(URUNTIME_XXHASH_LIBS ${XXHASH_LIBRARIES})
if(URUNTIME_STATIC)
    if(ZSTD_STATIC)
        set(URUNTIME_ZSTD_LIBS ${ZSTD_STATIC})
    endif()
    if(XXHASH_STATIC)
        set(URUNTIME_XXHASH_LIBS ${XXHASH_STATIC})
    endif()
endif()

# List of tool binaries to embed
set(EMBED_TOOLS "")
if(URUNTIME_SQUASHFS)
    list(APPEND EMBED_TOOLS squashfuse unsquashfs)
endif()
if(URUNTIME_SQUASHFS AND NOT URUNTIME_LITE)
    list(APPEND EMBED_TOOLS mksquashfs)
endif()
if(URUNTIME_DWARFS)
    list(APPEND EMBED_TOOLS dwarfs-universal dwarfsextract)
endif()

set(URUNTIME_EMBED_HEADERS "")
set(URUNTIME_EMBED_TOOLS "")

foreach(tool ${EMBED_TOOLS})
    string(REGEX REPLACE "-" "_" var_prefix ${tool})
    set(input "${CMAKE_CURRENT_SOURCE_DIR}/assets/${tool}-zst")
    if(NOT EXISTS "${input}")
        message(WARNING "Asset ${input} not found; skipping ${tool}")
        continue()
    endif()
    set(output "${CMAKE_CURRENT_BINARY_DIR}/${tool}_embedded.h")
    add_custom_command(
        OUTPUT  "${output}"
        COMMAND embed_compress "${input}" "${output}" "${var_prefix}_data"
        DEPENDS "${input}" embed_compress
        COMMENT "Embedding ${tool}..."
    )
    list(APPEND URUNTIME_EMBED_HEADERS "${output}")
    list(APPEND URUNTIME_EMBED_TOOLS "${tool}")
endforeach()

# Generate embed_config.h with HAVE_ macros for each embedded tool
set(EMBED_CONFIG_H "${CMAKE_CURRENT_BINARY_DIR}/embed_config.h")
file(WRITE "${EMBED_CONFIG_H}"
    "/* Auto-generated by CMake */\n"
    "#pragma once\n"
)
foreach(tool ${URUNTIME_EMBED_TOOLS})
    string(REGEX REPLACE "-" "_" var_prefix ${tool})
    file(APPEND "${EMBED_CONFIG_H}" "#define HAVE_${var_prefix}_DATA 1\n")
endforeach()

# Generate embed_data.c when tools are available
set(EMBED_DATA_C "")
if(URUNTIME_EMBED_HEADERS)
    set(EMBED_DATA_C "${CMAKE_CURRENT_BINARY_DIR}/embed_data.c")
    file(WRITE "${EMBED_DATA_C}"
        "/* Auto-generated by CMake - embedded data includes */\n"
        "#include <stddef.h>\n"
        "#include \"embed_config.h\"\n"
    )
    foreach(hdr ${URUNTIME_EMBED_HEADERS})
        file(RELATIVE_PATH rel "${CMAKE_CURRENT_BINARY_DIR}" "${hdr}")
        file(APPEND "${EMBED_DATA_C}" "#include \"${rel}\"\n")
    endforeach()
    file(APPEND "${EMBED_DATA_C}" "\n")
endif()

if(NOT URUNTIME_EMBED_HEADERS)
    message(WARNING "No embedded tool assets found. Building bare uruntime without SquashFS/DwarFS support. "
        "Run 'fetch_assets.sh <arch>' to add tools.")
endif()

# Main sources
set(URUNTIME_SOURCES
    src/main.c
    src/util.c
    src/elf.c
    src/embed.c
    src/ns.c
    src/signal.c
)
if(EMBED_DATA_C)
    list(APPEND URUNTIME_SOURCES "${EMBED_DATA_C}")
endif()

add_executable(uruntime ${URUNTIME_SOURCES})

if(URUNTIME_EMBED_HEADERS)
    add_custom_target(gen_embed_headers DEPENDS ${URUNTIME_EMBED_HEADERS})
    add_dependencies(uruntime gen_embed_headers)
endif()
if(EMBED_DATA_C)
    add_dependencies(uruntime embed_compress)
endif()

# Only use -static when zstd static lib is available (required dependency).
# Without it the link will fail; fall back gracefully on arches lacking static libs.
if(URUNTIME_STATIC AND ZSTD_STATIC)
    target_link_options(uruntime PRIVATE -static)
elseif(URUNTIME_STATIC AND NOT ZSTD_STATIC)
    message(WARNING "zstd static library not found, linking dynamically")
endif()

target_link_libraries(uruntime PRIVATE
    ${URUNTIME_ZSTD_LIBS}
    ${URUNTIME_XXHASH_LIBS}
    m
)

target_include_directories(uruntime PRIVATE
    ${ZSTD_INCLUDE_DIRS}
    ${XXHASH_INCLUDE_DIRS}
    "${CMAKE_CURRENT_BINARY_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
)

target_compile_definitions(uruntime PRIVATE
    _GNU_SOURCE
    URUNTIME_NAME="${PROJECT_NAME}"
    URUNTIME_VERSION="${PROJECT_VERSION}"
    URUNTIME_REPOSITORY="https://github.com/VHSgunzo/uruntime"
    $<$<BOOL:${URUNTIME_APPIMAGE}>:URUNTIME_APPIMAGE>
    $<$<BOOL:${URUNTIME_SQUASHFS}>:URUNTIME_SQUASHFS>
    $<$<BOOL:${URUNTIME_DWARFS}>:URUNTIME_DWARFS>
    $<$<BOOL:${URUNTIME_LITE}>:URUNTIME_LITE>
    $<$<BOOL:${URUNTIME_UPX}>:URUNTIME_UPX>
)
